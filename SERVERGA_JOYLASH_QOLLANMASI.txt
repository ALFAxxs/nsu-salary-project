========================================================================
NSU SALARY BOT — SERVERNI 0 DAN SOZLASH VA LOYIHANI ISHGA TUSHURISH
========================================================================
Bu qo'llanma toza Ubuntu server (22.04 yoki 24.04 LTS, tavsiya etiladi)
uchun yozilgan. Barcha komandalarni SSH orqali serverga kirib, yuqoridan
pastga qarab, ketma-ket bajaring. "# " bilan boshlangan qatorlar izoh —
ularni terish shart emas.

Talab qilinadigan narsalar (oldindan tayyorlab qo'ying):
  - Server (VPS): kamida 2 GB RAM, 1-2 CPU, 20+ GB disk
  - Domen nomi (masalan: oylik.example.uz) — DNS A-yozuvi shu serverning
    IP-manziliga yo'naltirilgan bo'lishi kerak
  - Sizning GitHub repo manzilingiz: github.com/ALFAxxs/nsu-salary-project
  - Telegram bot tokeni (@BotFather orqali olingan)


========================================================================

3-QADAM. POSTGRESQL O'RNATISH VA BAZA YARATISH
========================================================================
apt install -y postgresql postgresql-contrib

sudo -u postgres psql <<'EOF'
CREATE DATABASE salary_db;
CREATE USER salary_user WITH PASSWORD 'KUCHLI_PAROL_BU_YERGA';
ALTER ROLE salary_user SET client_encoding TO 'utf8';
ALTER ROLE salary_user SET default_transaction_isolation TO 'read committed';
ALTER ROLE salary_user SET timezone TO 'Asia/Tashkent';
GRANT ALL PRIVILEGES ON DATABASE salary_db TO salary_user;
\c salary_db
GRANT ALL ON SCHEMA public TO salary_user;
ALTER SCHEMA public OWNER TO salary_user;
\q
EOF
# MUHIM: PostgreSQL 15+ da "public" schemaga CREATE huquqi endi
# avtomatik berilmaydi (eski versiyalarda beriladi edi) — shuning
# uchun yuqoridagi ikkita GRANT/ALTER qatori shart, aks holda
# "python manage.py migrate" "permission denied for schema public"
# xatosi bilan to'xtaydi.

# ESLATMA: "KUCHLI_PAROL_BU_YERGA" o'rniga haqiqiy kuchli parol qo'ying
# va uni keyinroq .env fayliga aynan shu holda yozasiz.

10-QADAM. NGINX SOZLASH (avval SSL'siz, keyin SSL bilan)
========================================================================
# 10.1 — nginx.conf faylida domeningizni to'g'irlang:
nano /opt/salary_system/deploy/nginx.conf
# "example.com" so'zini HAR IKKALA joyda ham o'zingizning haqiqiy
# domeningizga almashtiring (masalan: oylik.example.uz)

# 10.2 — SSL sertifikat hali yo'q, shuning uchun faylni VAQTINCHA
# faqat 80-port (HTTP) bilan ishlaydigan qilib qo'yamiz. Faylning
# ikkinchi (443 ssl) blokini vaqtincha butunlay o'chirib turing —
# eng oson yo'li: 443-blokni "#" bilan izohga olish, YOKI birinchi
# blokdagi "return 301 https://..." qatorini vaqtincha
# "proxy_pass http://unix:/run/salary/salary.sock;" ga almashtirib turish.
#
# ENG ODDIY YECHIM — vaqtinchalik minimal config yozing:
cat > /etc/nginx/sites-available/salary <<'EOF'
server {
    listen 80;
    server_name oylik.example.uz www.oylik.example.uz;

    client_max_body_size 12M;

    location /static/ {
        alias /opt/salary_system/staticfiles/;
        expires 30d;
    }
    location / {
        proxy_pass http://unix:/run/salary/salary.sock;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
EOF
# (yuqoridagi "oylik.example.uz" larni o'zingiznikiga almashtirib qo'ying)

ln -s /etc/nginx/sites-available/salary /etc/nginx/sites-enabled/salary
rm -f /etc/nginx/sites-enabled/default
nginx -t
systemctl restart nginx

# Endi brauzerda http://oylik.example.uz ochilishi kerak (login sahifasi)

11-QADAM. SSL SERTIFIKAT OLISH (Let's Encrypt / Certbot)

# So'ralganda: email kiriting, shartlarga rozilik bering,
# "Redirect HTTP to HTTPS" — HA (2-variant) ni tanlang.

# Certbot avtomatik ravishda nginx configni SSL bilan yangilaydi.
# Endi to'liq deploy/nginx.conf dagi variantga (media/internal,
# X-Forwarded-Proto va h.k.) mos kelishini tekshirib qo'yish uchun:
diff /opt/salary_system/deploy/nginx.conf /etc/nginx/sites-available/salary
# Farqlarni ko'rib, kerak bo'lsa /etc/nginx/sites-available/salary ni
# deploy/nginx.conf dagi to'liq nusxaga moslashtirib tahrirlang
# (ayniqsa /media/ location bloki — u boshlang'ich configda yo'q edi).
nginx -t && systemctl reload nginx

# .env faylida:
nano /opt/salary_system/.env
# SECURE_SSL_REDIRECT=True qilib qo'ying (agar hali qilmagan bo'lsangiz)
systemctl restart salary-web.service
